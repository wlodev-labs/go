package models

type GenericBadRequestResponseParams struct {
	Message *string `json:"message"`
}

func GenericBadRequestResponse(params GenericBadRequestResponseParams) BadRequestJSONResponse {
	var message *string
	if params.Message != nil {
		message = params.Message
	}

	return BadRequestJSONResponse{
		Error: BadRequestError{
			Code:    BadRequestErrorCodeBadRequest,
			Message: message,
		},
	}
}
